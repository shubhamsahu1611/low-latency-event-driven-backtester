cmake_minimum_required(VERSION 3.20)
project(low_latency_backtester VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

option(BT_ENABLE_AVX2 "Build runtime-dispatched AVX2 risk calculations on x86" ON)
option(BT_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(BT_TSAN "Enable ThreadSanitizer (separate build from BT_SANITIZE)" OFF)
if(BT_SANITIZE AND BT_TSAN)
    message(FATAL_ERROR "Use separate builds for ASan/UBSan and TSan")
endif()
if(BT_SANITIZE OR BT_TSAN)
    if(MSVC OR WIN32)
        message(FATAL_ERROR "Use GCC or Clang on Linux for the sanitizer options")
    endif()
    if(BT_TSAN)
        set(BT_SANITIZERS -fsanitize=thread)
    else()
        set(BT_SANITIZERS -fsanitize=address,undefined)
    endif()
    add_compile_options(${BT_SANITIZERS} -fno-omit-frame-pointer)
    add_link_options(${BT_SANITIZERS})
endif()

add_library(backtester_core src/risk.cpp src/data.cpp)
target_include_directories(backtester_core PUBLIC include)
target_link_libraries(backtester_core PUBLIC Threads::Threads)
if(BT_ENABLE_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i[3-6]86|x86)$")
    target_sources(backtester_core PRIVATE src/risk_avx2.cpp)
    target_compile_definitions(backtester_core PRIVATE BT_HAS_AVX2=1)
    if(MSVC)
        set_source_files_properties(src/risk_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    else()
        set_source_files_properties(src/risk_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-ffp-contract=off")
    endif()
endif()

add_executable(backtester src/main.cpp)
target_link_libraries(backtester PRIVATE backtester_core)
target_compile_definitions(backtester PRIVATE
    BT_BENCHMARK_BUILD=$<IF:$<CONFIG:Release>,1,0>
    BT_BUILD_TYPE="$<CONFIG>")

include(CTest)
if(BUILD_TESTING)
    add_executable(backtester_tests tests/test_main.cpp tests/test_queue_pool.cpp
        tests/test_trading.cpp tests/test_risk.cpp tests/test_data.cpp)
    target_link_libraries(backtester_tests PRIVATE backtester_core)
    target_compile_definitions(backtester_tests PRIVATE BT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME components COMMAND backtester_tests)
    set_tests_properties(components PROPERTIES TIMEOUT 60)
    add_test(NAME replay COMMAND backtester replay --csv ${CMAKE_CURRENT_SOURCE_DIR}/data/example.csv --runs 1 --strategy-every 1)
    add_test(NAME invalid_cli COMMAND backtester benchmark --ticks -1)
    set_tests_properties(invalid_cli PROPERTIES WILL_FAIL TRUE)
endif()

set(BT_TARGETS backtester_core backtester)
if(BUILD_TESTING)
    list(APPEND BT_TARGETS backtester_tests)
endif()
foreach(target IN LISTS BT_TARGETS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    endif()
endforeach()
